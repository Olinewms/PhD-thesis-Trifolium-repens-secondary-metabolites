# Note that braker already does this section for you (braker.aa), but you could use this in situations where you don't have such a fasta already available. 
# Extract CDS's using gff3 and reference genome
gffread -g ../braker_working_dir/UTM_Trep_v1.0_hap1_genomic.fa -x braker_cds_sequences.fasta ../braker_working_dir/braker/braker.gff3

# Translate all CDS's
transeq -sequence braker_cds_sequences.fasta -outseq braker_protein_sequences.fasta

awk -F '[|]' '{if ($2 ~ /^g[0-9]+$/) print $2}' filtered_annotated_variants.tsv | sort -u > gene_list.txt

awk 'BEGIN { while ((getline < "gene_list.txt") > 0) genes[$1] = 1 } 
     /^>/ {match($0, /gene=([^ ]+)/, arr); keep = (arr[1] in genes)} 
     keep' braker_protein_sequences.fasta > filtered_proteins.fasta
##################################################################################################################################################################


# This removes rows that contain ".", "@", "intron" and "synonymous" in the 5th column
# This was done so we were left with only nonsynonymous mutations in the CDS's
awk '$5 !~ /^(\.|@|intron|synonymous)/' merged_csq_annotated_variants_LowVsHigh.tsv > filtered_annotated_variants_nonsynonymous_excluding_UTRs_and_introns.tsv

# Then make a list of all the transcripts that contain nonsynonymous mutations
cut -f5 filtered_annotated_variants_nonsynonymous_excluding_UTRs_and_introns.tsv | awk -F'|' '{print $3}' > filtered_annotated_variants_transcript_ids.txt

# Then use this list to pull out the protein sequences from the braker.aa fasta
# The following bash script is saved as extract_transcripts.sh

#!/bin/bash
# Ensure the output file is empty before starting
> filtered_proteins.fasta

# Loop over each transcript ID in transcript_ids.txt
while IFS= read -r transcript; do
    # Extract the FASTA entry for the transcript.
    # This awk command prints the header and all sequence lines until the next header.
    awk -v id=">$transcript" '
      $0 ~ id {print_flag=1} 
      /^>/ && $0 !~ id {print_flag=0} 
      print_flag {print}
    ' braker.aa >> filtered_proteins.fasta
done < filtered_annotated_variants_transcript_ids.txt

# Then you can do a blastp nr search using filtered_proteins.fasta !!
